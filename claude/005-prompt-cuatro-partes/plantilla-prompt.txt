{
    var total = 0m;
    foreach (var item in pedido.Items)
        total += item.Precio * item.Cantidad;

    var resultado = _pasarela.Cobrar(numeroTarjeta, total);
    _logger.LogInformation("Cobrando " + total + " a la tarjeta " + numeroTarjeta);

    if (resultado == "OK")
